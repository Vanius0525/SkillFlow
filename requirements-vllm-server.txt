# vLLM inference server — install on the Linux 4090 box (driver 570.x / CUDA 12.8)
#
#   uv venv ~/vllm-env --python 3.12
#   source ~/vllm-env/bin/activate
#   uv pip install -r requirements-vllm-server.txt --torch-backend=cu128
#
# Notes:
# - Do NOT install torch separately; vllm pins its own torch (2.8.x, cu128 wheels),
#   which runs fine on a CUDA 12.8 driver. `--torch-backend=cu128` forces uv to
#   resolve torch from the cu128 index and never pick a cu130 (CUDA 13) build.
# - vllm 0.11.x default PyPI wheels are built against CUDA 12.8 — no source build needed.
